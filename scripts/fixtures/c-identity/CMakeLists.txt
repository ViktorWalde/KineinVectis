cmake_minimum_required(VERSION 3.25)
project(c_identity LANGUAGES C CXX)
add_library(c_api STATIC src/c_api.c include/c_api.h include/shared.h)
target_include_directories(c_api PUBLIC include)
set_target_properties(c_api PROPERTIES C_STANDARD 23 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
add_executable(mixed src/main.cpp include/cpp_api.h include/shared.h)
target_link_libraries(mixed PRIVATE c_api)
set_target_properties(mixed PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
enable_testing()
add_test(NAME mixed_api COMMAND mixed)
